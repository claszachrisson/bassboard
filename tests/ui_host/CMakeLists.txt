cmake_minimum_required(VERSION 3.16)
project(home_test LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(home_test home_test.cpp ../../main/src/ui/ui.cpp ../../main/src/ui/fonts/card_bold_24.c ../../main/src/ui/fonts/card_bold_20.c ../../main/src/ui/fonts/card_bold_16.c)
target_include_directories(home_test PRIVATE ../../main/src/ui ../../main/src/core)
target_link_libraries(home_test PRIVATE lvgl)
